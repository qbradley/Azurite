5. KEY BEHAVIORAL DIFFERENCES: TS → RUST
   ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
   
   TS: for(auth of auths) { pass = await auth.validate(); if(pass) return true; }
   Rust: for auth in auths { if let Some(pass) = auth.validate().await? { return Ok(()); } }
   
   TS: .then().catch(next) — error handler pattern
   Rust: .await? or match/if-let — Result/Option handling
   
   TS: undefined/false both mean failure
   Rust: Option<bool> — None=skip, Some(true)=pass, Some(false)=fail
   
   TS: res.locals[path][key] = value
   Rust: context.set_string(key, value) with Arc<Mutex<>> interior mutability
   
   TS: JSON/Map objects for CORS rules
   Rust: GeneratedObject with field extraction via helpers
